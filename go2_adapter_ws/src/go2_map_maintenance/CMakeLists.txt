cmake_minimum_required(VERSION 3.8)
project(go2_map_maintenance)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/MapSnapshot.msg" DEPENDENCIES std_msgs sensor_msgs)
add_executable(map_maintenance_node src/map_maintenance_node.cpp)
target_include_directories(map_maintenance_node PRIVATE include)
target_compile_options(map_maintenance_node PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(map_maintenance_node rclcpp nav_msgs sensor_msgs std_msgs)
rosidl_target_interfaces(map_maintenance_node ${PROJECT_NAME} "rosidl_typesupport_cpp")
install(TARGETS map_maintenance_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_voxel_map test/test_voxel_map.cpp)
  target_include_directories(test_voxel_map PRIVATE include)
  add_test(NAME test_voxel_map COMMAND test_voxel_map)
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
